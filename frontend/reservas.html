const API = "http://127.0.0.1:5000"; async function carregar() { const token = localStorage.getItem("token"); if (!token) { window.location.href = "index.html"; return; } try { const resposta = await fetch( `${API}/reservas`, { headers: { "Authorization": token } } ); const reservas = await resposta.json(); if (!resposta.ok) { alert(reservas.mensagem); return; } const lista = document.getElementById("lista"); lista.innerHTML = ""; if (reservas.length === 0) { lista.innerHTML = "

Nenhuma reserva encontrada.

"; return; } reservas.forEach(function(r) { lista.innerHTML += `

Ambiente: ${r.ambiente}

Matéria: ${r.materia}

Data: ${r.data}

Horário: ${r.hora}


`; }); } catch (erro) { console.error(erro); alert( "Erro ao carregar as reservas." ); } } carregar();
